These boxplots compare the distribution of one EEG biomarker at a time between patients with mild cognitive impairment (Group MCI) and Alzheimer’s disease (Group AD) after filtering by age and gender. Key takeaways from correlation heatmap

These correlation heatmap help identify clusters of biomarkers that move together and may therefore contain overlapping information. Strong positive or negative correlations suggest redundancy, while weaker correlations highlight features that may contribute unique information to downstream modeling or interpretation.

The filters let the viewer test whether the correlation structure changes across diagnosis group, gender, age range, feature family, and correlation method. This supports deeper exploration by allowing the user to move from a broad overview of the full biomarker set to a focused view of only the strongest relationships.

to_display <- function(clean_names_vec) {
  out <- name_map$display[match(clean_names_vec, name_map$clean)]
  ifelse(is.na(out), clean_names_vec, out)
}

to_clean <- function(display_name) {
  out <- name_map$clean[match(display_name, name_map$display)]
  ifelse(is.na(out), display_name, out)
}

selected_features_by_family <- function(families) {
  feature_family %>%
    filter(family %in% families) %>%
    pull(feature)
}

calc_p_value <- function(dat, feature_col) {
  tmp <- dat %>%
    select(diagnosis, all_of(feature_col)) %>%
    drop_na()

  if (nrow(tmp) == 0) return(NA_real_)
  if (length(unique(tmp$diagnosis)) != 2) return(NA_real_)

  tryCatch(
    wilcox.test(tmp[[feature_col]] ~ tmp$diagnosis)$p.value,
    error = function(e) NA_real_
  )
}

make_corr <- function(dat, features, method = "pearson") {
  X <- dat %>%
    select(all_of(features)) %>%
    mutate(across(everything(), as.numeric))

  # Keep columns with at least 3 non-missing values and non-zero variance
  keep_cols <- names(X)[sapply(X, function(x) sum(!is.na(x)) >= 3 && sd(x, na.rm = TRUE) > 0)]
  X <- X %>% select(all_of(keep_cols))

  if (ncol(X) < 2) return(NULL)

  cor(X, use = "pairwise.complete.obs", method = method)
}

label_lookup <- function(x) {
  ifelse(x %in% names(simple_labels), unname(simple_labels[x]), x)
}

```{r render-boxplot}
feature_col <- "delta_abs"
feature_label <- "log10 absolute delta power"


plot_dat <- patient_result_clean %>%
  select(patient_id, diagnosis, gender, age, all_of(feature_col)) %>%
  mutate(
    raw_value = .data[[feature_col]],
    value_plot = log10(raw_value + 1)
  )

p_val <- wilcox.test(value_plot ~ diagnosis, data = plot_dat)$p.value



p <- ggplot(
  plot_dat,
  aes(
    x = diagnosis,
    y = value_plot,
    fill = diagnosis,
    text = paste0(
      "Patient ID: ", patient_id,
      "<br>Diagnosis: ", diagnosis,
      "<br>Age: ", age,
      "<br>Value: ", round(raw_value,3)
    )
  )
) +
  geom_boxplot(width = 0.6, color = "black") +
  geom_jitter(width = 0.1, size = 2, alpha = 0.6) +
  scale_fill_manual(values = c("MCI"="#2C7BB6","AD"="#D7191C")) +
  labs(
      x = NULL,
      y = NULL,
      fill = NULL
    )+
  theme_bw(base_size = 13) +
  theme(
    legend.position = "bottom",
    panel.grid.major.x = element_blank(),
    panel.grid.minor = element_blank(),
    axis.text = element_text(size = 14),
    axis.title = element_text(size = 15),
    plot.title = element_text(face = "bold", size = 16, hjust = 0.5),
    plot.subtitle = element_text(size = 11)
  )

ggplotly(p, tooltip="text") %>%
  layout(
    title = list(
      text = paste0(
        "Distribution of ",feature_label,
        "<br><span style='font-size:14px'>Wilcoxon p-value: ",
        signif(p_val,3),
        "</span>"
      ),
      x = 0.5
    ),
    margin=list(t=120)
  )
```
